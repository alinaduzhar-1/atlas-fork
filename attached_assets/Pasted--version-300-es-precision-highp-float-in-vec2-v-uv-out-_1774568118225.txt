#version 300 es
precision highp float;

in vec2 v_uv;
out vec4 fragColor;

uniform float u_time;
uniform vec2 u_resolution;
uniform float u_intensity;
uniform float u_symmetry;
uniform float u_noiseScale;
uniform float u_noiseSpeed;
uniform float u_grainAmount;
uniform float u_colorSpread;
uniform vec3 u_primaryColor;
uniform vec3 u_secondaryColor;
uniform vec3 u_accentColor;
uniform vec3 u_extraColor;

vec3 mod289(vec3 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
vec4 mod289(vec4 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
vec4 permute(vec4 x) { return mod289(((x * 34.0) + 10.0) * x); }
vec4 taylorInvSqrt(vec4 r) { return 1.79284291400159 - 0.85373472095314 * r; }

float snoise(vec3 v) {
  const vec2 C = vec2(1.0 / 6.0, 1.0 / 3.0);
  const vec4 D = vec4(0.0, 0.5, 1.0, 2.0);

  vec3 i = floor(v + dot(v, C.yyy));
  vec3 x0 = v - i + dot(i, C.xxx);

  vec3 g = step(x0.yzx, x0.xyz);
  vec3 l = 1.0 - g;
  vec3 i1 = min(g.xyz, l.zxy);
  vec3 i2 = max(g.xyz, l.zxy);

  vec3 x1 = x0 - i1 + C.xxx;
  vec3 x2 = x0 - i2 + C.yyy;
  vec3 x3 = x0 - D.yyy;

  i = mod289(i);
  vec4 p = permute(permute(permute(
    i.z + vec4(0.0, i1.z, i2.z, 1.0))
    + i.y + vec4(0.0, i1.y, i2.y, 1.0))
    + i.x + vec4(0.0, i1.x, i2.x, 1.0));

  float n_ = 0.142857142857;
  vec3 ns = n_ * D.wyz - D.xzx;

  vec4 j = p - 49.0 * floor(p * ns.z * ns.z);

  vec4 x_ = floor(j * ns.z);
  vec4 y_ = floor(j - 7.0 * x_);

  vec4 x = x_ * ns.x + ns.yyyy;
  vec4 y = y_ * ns.x + ns.yyyy;
  vec4 h = 1.0 - abs(x) - abs(y);

  vec4 b0 = vec4(x.xy, y.xy);
  vec4 b1 = vec4(x.zw, y.zw);

  vec4 s0 = floor(b0) * 2.0 + 1.0;
  vec4 s1 = floor(b1) * 2.0 + 1.0;
  vec4 sh = -step(h, vec4(0.0));

  vec4 a0 = b0.xzyw + s0.xzyw * sh.xxyy;
  vec4 a1 = b1.xzyw + s1.xzyw * sh.zzww;

  vec3 p0 = vec3(a0.xy, h.x);
  vec3 p1 = vec3(a0.zw, h.y);
  vec3 p2 = vec3(a1.xy, h.z);
  vec3 p3 = vec3(a1.zw, h.w);

  vec4 norm = taylorInvSqrt(vec4(dot(p0, p0), dot(p1, p1), dot(p2, p2), dot(p3, p3)));
  p0 *= norm.x;
  p1 *= norm.y;
  p2 *= norm.z;
  p3 *= norm.w;

  vec4 m = max(0.6 - vec4(dot(x0, x0), dot(x1, x1), dot(x2, x2), dot(x3, x3)), 0.0);
  m = m * m;
  return 42.0 * dot(m * m, vec4(dot(p0, x0), dot(p1, x1), dot(p2, x2), dot(p3, x3)));
}

float fbm(vec3 p, int octaves) {
  float value = 0.0;
  float amplitude = 0.5;
  float frequency = 1.0;
  for (int i = 0; i < 6; i++) {
    if (i >= octaves) break;
    value += amplitude * snoise(p * frequency);
    frequency *= 2.0;
    amplitude *= 0.5;
  }
  return value;
}

float grain(vec2 uv, float t) {
  return fract(sin(dot(uv + t, vec2(12.9898, 78.233))) * 43758.5453);
}

void main() {
  vec2 uv = v_uv;
  float aspect = u_resolution.x / u_resolution.y;
  vec2 centeredUv = (uv - 0.5) * vec2(aspect, 1.0);

  vec2 symmetricUv = centeredUv;
  symmetricUv.x = mix(centeredUv.x, abs(centeredUv.x), u_symmetry);

  float time = u_time * u_noiseSpeed;

  float n1 = fbm(vec3(symmetricUv * u_noiseScale, time), 4);
  float n2 = fbm(vec3(symmetricUv * u_noiseScale * 1.5 + 3.7, time * 0.7), 3);
  float n3 = fbm(vec3(symmetricUv * u_noiseScale * 0.8 - 2.1, time * 1.3), 5);

  float wingShape = 1.0 - smoothstep(0.0, 0.8, length(centeredUv * vec2(0.7, 1.0)));
  float wingDetail = snoise(vec3(symmetricUv * u_noiseScale * 2.0, time * 0.5));
  wingShape *= mix(0.5, 1.0, wingDetail * 0.5 + 0.5);

  float colorMix1 = smoothstep(-u_colorSpread, u_colorSpread, n1);
  float colorMix2 = smoothstep(-u_colorSpread, u_colorSpread, n2);
  float colorMix3 = smoothstep(-u_colorSpread * 0.5, u_colorSpread * 0.5, n3);

  vec3 color = mix(u_primaryColor, u_secondaryColor, colorMix1);
  color = mix(color, u_accentColor, colorMix2 * u_intensity * 0.7);
  color = mix(color, u_extraColor, colorMix3 * u_intensity * 0.5);

  color = mix(color * 0.8, color, wingShape * u_intensity);

  float contrast = 1.0 + u_intensity * 0.5;
  color = pow(color, vec3(contrast));
  color = mix(vec3(0.5), color, 0.8 + u_intensity * 0.4);

  float g = grain(uv * u_resolution, fract(u_time * 0.1)) * u_grainAmount;
  color += vec3(g) - u_grainAmount * 0.5;

  color = mix(vec3(0.02, 0.02, 0.03), color, 0.15 + wingShape * 0.85 * u_intensity);

  color = clamp(color, 0.0, 1.0);

  fragColor = vec4(color, 1.0);
}
